# master-ai — План 2: Веб-UI и редактор

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Рабочий веб-интерфейс: главная со списком кампаний, страница кампании с дашбордом документов по типам, и редактор документа на TipTap (без AI) с сохранением через REST API Плана 1.

**Architecture:** Новый пакет `@master-ai/web` (Vite + React + TS). Общается с бэкендом (`@master-ai/server`, порт 5174) через типизированный fetch-клиент; в dev Vite проксирует `/api` на бэкенд. Роутинг — react-router. Состояние — локальный React state + useEffect (без сторонних data-библиотек). Типы документов/кампаний переиспользуются из `@master-ai/shared`.

**Tech Stack:** Vite 5, React 18, react-router-dom 6, @tiptap/react + @tiptap/starter-kit, Vitest + @testing-library/react + jsdom.

## Global Constraints

- Node.js 20+, pnpm workspaces. TypeScript ESM.
- Тесты — Vitest (+ jsdom для компонентов). `typecheck` через `tsc --noEmit` должен проходить (как в Плане 1).
- Каждый таск завершается зелёными тестами + коммит. Коммиты — только имя автора, без Co-Authored-By/AI-трейлеров.
- API бэкенда (готов в Плане 1, порт 5174): `GET /api/document-types`, `GET /api/campaigns`, `POST /api/campaigns` (body `{name, system?, description?}` → 201), `GET /api/campaigns/:id` (404), `GET /api/campaigns/:id/documents`, `POST /api/documents` (body `{campaignId, type, title, status?, contentJson?, fields?}` → 201; 400 на неизвестный type/невалидный status; 404 на несуществующий campaignId), `GET /api/documents/:id` (404), `PATCH /api/documents/:id` (body — частичный `{title?, status?, contentJson?, fields?}` → 404 если нет).
- Типы из `@master-ai/shared`: `Campaign`, `NewCampaign`, `Document`, `NewDocument`, `DocumentTypeDef`, `DOCUMENT_TYPES`, `getDocumentType`.
- Рабочая директория: `Docs/НРИ/master-ai/`.

`packages/web/src/api/client.ts`:
```ts
import type {
  Campaign,
  NewCampaign,
  Document,
  NewDocument,
  DocumentTypeDef,
} from "@master-ai/shared";

export class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
    this.name = "ApiError";
  }
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, {
    ...init,
    headers: { "content-type": "application/json", ...init?.headers },
  });
  if (!res.ok) {
    let message = res.statusText;
    try {
      const body = (await res.json()) as { error?: string };
      if (body.error) message = body.error;
    } catch {
      // тело не JSON — оставляем statusText
    }
    throw new ApiError(res.status, message);
  }
  return (await res.json()) as T;
}

type DocumentPatch = Partial<Pick<Document, "title" | "status" | "contentJson" | "fields">>;

export const api = {
  listDocumentTypes: () => request<DocumentTypeDef[]>("/api/document-types"),
  listCampaigns: () => request<Campaign[]>("/api/campaigns"),
  createCampaign: (input: NewCampaign) =>
    request<Campaign>("/api/campaigns", { method: "POST", body: JSON.stringify(input) }),
  getCampaign: (id: string) => request<Campaign>(`/api/campaigns/${id}`),
  listCampaignDocuments: (campaignId: string) =>
    request<Document[]>(`/api/campaigns/${campaignId}/documents`),
  createDocument: (input: NewDocument) =>
    request<Document>("/api/documents", { method: "POST", body: JSON.stringify(input) }),
  getDocument: (id: string) => request<Document>(`/api/documents/${id}`),
  updateDocument: (id: string, patch: DocumentPatch) =>
    request<Document>(`/api/documents/${id}`, { method: "PATCH", body: JSON.stringify(patch) }),
};
```

import { useEffect, useState, type FormEvent } from "react";
import { Link } from "react-router-dom";
import type { Campaign } from "@master-ai/shared";
import { api } from "../api/client.js";

export function HomePage() {
  const [campaigns, setCampaigns] = useState<Campaign[]>([]);
  const [name, setName] = useState("");

  useEffect(() => {
    api.listCampaigns().then(setCampaigns);
  }, []);

  async function handleSubmit(e: FormEvent) {
    e.preventDefault();
    if (!name.trim()) return;
    const created = await api.createCampaign({ name: name.trim() });
    setCampaigns((prev) => [...prev, created]);
    setName("");
  }

  return (
    <main>
      <h1>master-ai</h1>
      <h2>Кампании</h2>
      <ul>
        {campaigns.map((c) => (
          <li key={c.id}>
            <Link to={`/campaigns/${c.id}`}>{c.name}</Link>
          </li>
        ))}
      </ul>
      <form onSubmit={handleSubmit}>
        <label htmlFor="campaign-name">Название</label>
        <input
          id="campaign-name"
          value={name}
          onChange={(e) => setName(e.target.value)}
        />
        <button type="submit">Создать</button>
      </form>
    </main>
  );
}

`packages/web/src/pages/CampaignPage.tsx`:
```tsx
import { useEffect, useState } from "react";
import { Link, useNavigate, useParams } from "react-router-dom";
import {
  DOCUMENT_TYPES,
  getDocumentType,
  type Campaign,
  type Document,
} from "@master-ai/shared";
import { api } from "../api/client.js";

export function CampaignPage() {
  const { campaignId } = useParams<{ campaignId: string }>();
  const navigate = useNavigate();
  const [campaign, setCampaign] = useState<Campaign | null>(null);
  const [documents, setDocuments] = useState<Document[]>([]);
  const [type, setType] = useState(DOCUMENT_TYPES[0].key);

  useEffect(() => {
    if (!campaignId) return;
    api.getCampaign(campaignId).then(setCampaign);
    api.listCampaignDocuments(campaignId).then(setDocuments);
  }, [campaignId]);

  async function createDocument() {
    if (!campaignId) return;
    const doc = await api.createDocument({ campaignId, type, title: "Без названия" });
    navigate(`/documents/${doc.id}`);
  }

  const byType = DOCUMENT_TYPES.map((t) => ({
    type: t,
    docs: documents.filter((d) => d.type === t.key),
  })).filter((g) => g.docs.length > 0);

  return (
    <main>
      <p><Link to="/">← Кампании</Link></p>
      <h1>{campaign?.name ?? "…"}</h1>

      <div>
        <select value={type} onChange={(e) => setType(e.target.value)} aria-label="Тип документа">
          {DOCUMENT_TYPES.map((t) => (
            <option key={t.key} value={t.key}>{t.label}</option>
          ))}
        </select>
        <button type="button" onClick={createDocument}>Новый документ</button>
      </div>

      {byType.map((group) => (
        <section key={group.type.key}>
          <h2>{group.type.label}</h2>
          <ul>
            {group.docs.map((d) => (
              <li key={d.id}>
                <Link to={`/documents/${d.id}`}>{d.title}</Link>
              </li>
            ))}
          </ul>
        </section>
      ))}
    </main>
  );
}
```

> `getDocumentType` импортирован для согласованности с реестром, но в этом компоненте группировка идёт по `DOCUMENT_TYPES` напрямую; если линтер ругается на неиспользуемый импорт — убери `getDocumentType` из импорта.

- [ ] **Step 3: Реализовать DocumentEditor**

`packages/web/src/editor/DocumentEditor.tsx`:
```tsx
import { useEditor, EditorContent } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";

interface DocumentEditorProps {
  content: unknown;
  onChange: (json: unknown) => void;
}

export function DocumentEditor({ content, onChange }: DocumentEditorProps) {
  const editor = useEditor({
    extensions: [StarterKit],
    content: (content as object) ?? "",
    onUpdate: ({ editor }) => onChange(editor.getJSON()),
  });

  return <EditorContent editor={editor} />;
}
```

> Важно: `content` используется только для инициализации редактора. Смена документа обрабатывается через `key={documentId}` на стороне `DocumentPage` (ремоунт редактора с новым начальным содержимым) — поэтому здесь НЕТ `useEffect`, который сбрасывал бы контент на каждое нажатие (это ломало бы курсор при наборе, т.к. `DocumentPage` поднимает состояние `contentJson` и обновляет его на каждый `onChange`).

- [ ] **Step 4: Реализовать DocumentPage**

`packages/web/src/pages/DocumentPage.tsx`:
```tsx
import { useEffect, useState } from "react";
import { Link, useParams } from "react-router-dom";
import { getDocumentType, type Document } from "@master-ai/shared";
import { api } from "../api/client.js";
import { DocumentEditor } from "../editor/DocumentEditor.js";

export function DocumentPage() {
  const { documentId } = useParams<{ documentId: string }>();
  const [doc, setDoc] = useState<Document | null>(null);
  const [title, setTitle] = useState("");
  const [status, setStatus] = useState<string>("");
  const [contentJson, setContentJson] = useState<unknown>(null);

  useEffect(() => {
    if (!documentId) return;
    api.getDocument(documentId).then((d) => {
      setDoc(d);
      setTitle(d.title);
      setStatus(d.status ?? "");
      setContentJson(d.contentJson);
    });
  }, [documentId]);

  async function save() {
    if (!documentId) return;
    const updated = await api.updateDocument(documentId, {
      title,
      status: status === "" ? null : status,
      contentJson,
    });
    setDoc(updated);
  }

  if (!doc) return <main>…</main>;

  const statuses = getDocumentType(doc.type)?.statuses ?? [];

  return (
    <main>
      <p><Link to={`/campaigns/${doc.campaignId}`}>← Кампания</Link></p>

      <label htmlFor="doc-title">Заголовок</label>
      <input id="doc-title" value={title} onChange={(e) => setTitle(e.target.value)} />

      {statuses.length > 0 && (
        <>
          <label htmlFor="doc-status">Статус</label>
          <select id="doc-status" value={status} onChange={(e) => setStatus(e.target.value)}>
            <option value="">—</option>
            {statuses.map((s) => (
              <option key={s} value={s}>{s}</option>
            ))}
          </select>
        </>
      )}

      <DocumentEditor key={documentId} content={contentJson} onChange={setContentJson} />

      <button type="button" onClick={save}>Сохранить</button>
    </main>
  );
}
```
